cmake_minimum_required(VERSION 3.20)
project(limit_order_book VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(LOB_BUILD_TESTS "Build Google Test correctness tests" ON)
option(LOB_FETCH_GTEST "Download Google Test if not installed" ON)
option(LOB_ASAN_UBSAN "Enable address and undefined behavior sanitizers" OFF)
option(LOB_UBSAN "Enable only undefined behavior sanitizer" OFF)
option(LOB_TSAN "Enable thread sanitizer (separate build)" OFF)
if((LOB_ASAN_UBSAN OR LOB_UBSAN) AND LOB_TSAN)
  message(FATAL_ERROR "ASan and TSan require separate builds")
endif()
find_package(Threads REQUIRED)
add_library(lob_options INTERFACE)
target_compile_options(lob_options INTERFACE
  $<$<CXX_COMPILER_ID:Clang,AppleClang,GNU>:-Wall;-Wextra;-Wpedantic;-Wconversion;-Wshadow>)
if(LOB_ASAN_UBSAN)
  target_compile_options(lob_options INTERFACE -fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
  target_link_options(lob_options INTERFACE -fsanitize=address,undefined)
endif()
if(LOB_UBSAN AND NOT LOB_ASAN_UBSAN)
  target_compile_options(lob_options INTERFACE -fsanitize=undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
  target_link_options(lob_options INTERFACE -fsanitize=undefined)
endif()
if(LOB_TSAN)
  target_compile_options(lob_options INTERFACE -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(lob_options INTERFACE -fsanitize=thread)
endif()
add_library(lob src/engine.cpp src/replay.cpp)
target_include_directories(lob PUBLIC include)
target_link_libraries(lob PUBLIC lob_options Threads::Threads)
add_executable(lob_sim apps/simulator.cpp)
target_link_libraries(lob_sim PRIVATE lob)
add_executable(lob_bench benchmarks/benchmark.cpp)
target_link_libraries(lob_bench PRIVATE lob)
if(LOB_BUILD_TESTS)
  enable_testing()
  find_package(GTest 1.15 CONFIG QUIET)
  if(NOT GTest_FOUND)
    if(NOT LOB_FETCH_GTEST)
      message(FATAL_ERROR "Install Google Test or enable LOB_FETCH_GTEST")
    endif()
    include(FetchContent)
    if(POLICY CMP0135)
      cmake_policy(SET CMP0135 NEW)
    endif()
    set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
    set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(googletest
      URL https://codeload.github.com/google/googletest/tar.gz/refs/tags/v1.15.2
      URL_HASH SHA256=7b42b4d6ed48810c5362c265a17faebe90dc2373c885e5216439d37927f02926
      TIMEOUT 45)
    FetchContent_MakeAvailable(googletest)
  endif()
  add_executable(lob_tests tests/engine_test.cpp tests/differential_test.cpp tests/queue_test.cpp tests/replay_test.cpp)
  target_link_libraries(lob_tests PRIVATE lob GTest::gtest_main)
  # Some preinstalled shared GTest packages omit their imported rpath metadata.
  set_property(TARGET lob_tests APPEND PROPERTY BUILD_RPATH
    "$<TARGET_FILE_DIR:GTest::gtest>;$<TARGET_FILE_DIR:GTest::gtest_main>")
  include(GoogleTest)
  gtest_discover_tests(lob_tests DISCOVERY_TIMEOUT 30 DISCOVERY_MODE PRE_TEST)
endif()
